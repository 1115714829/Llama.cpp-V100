=== 1cat-gpu-init service (what it sets up) === 
# /root/llm/systemd/1cat-gpu-init.service
[Unit]
Description=Prepare NVIDIA compute for AC922 1Cat after NFS root boot
Requires=nvidia-persistenced.service
After=remote-fs.target nvidia-persistenced.service
RequiresMountsFor=/root/llm/ac922env/1cat-20260907 /usr/local/cuda-12.4

[Service]
Type=oneshot
RemainAfterExit=yes
ExecStart=/usr/sbin/modprobe nvidia
ExecStart=/usr/sbin/modprobe nvidia_uvm
ExecStart=/usr/bin/nvidia-smi --query-gpu=index,name,driver_version --format=csv,noheader
TimeoutStartSec=120

[Install]
WantedBy=multi-user.target
DONE
